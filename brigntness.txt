#include <stdint.h>

/**
 * 核心缩放函数 (类似 FastLED scale8)
 * 确保 (color * brightness) 在 16 位空间计算，防止 8 位溢出导致的“乱跳”
 */
static inline uint8_t apply_brightness(uint8_t color, uint8_t brightness) {
    return (uint8_t)(((uint16_t)color * brightness) >> 8);
}

/**

/**
 * 最终 RGB 处理函数
 * @param r, g, b: 原始颜色分量
 * @param level: 期望亮度等级 (0-255)
 */
void process_rgb(uint8_t r, uint8_t g, uint8_t b, uint8_t level) {
    // 1. 视觉映射：将 0-255 的线性 level 映射为符合人眼的实际比例
    uint8_t visual_ratio = gamma8[level];

    // 2. 比例缩放：应用到每个通道
    uint8_t outR = apply_brightness(r, visual_ratio);
    uint8_t outG = apply_brightness(g, visual_ratio);
    uint8_t outB = apply_brightness(b, visual_ratio);
